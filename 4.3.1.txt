#include <iostream>
using namespace std;

int main() {
  int score;
  cout << "Введіть бал (1-100): ";
  cin >> score;

  if (score < 1 || score > 100)
    cout << "Некоректний бал\n";
  else if (score >= 90)
    cout << "A\n";
  else if (score >= 82)
    cout << "B\n";
  else if (score >= 74)
    cout << "C\n";
  else if (score >= 64)
    cout << "D\n";
  else if (score >= 60)
    cout << "E\n";
  else if (score >= 35)
    cout << "FX\n";
  else
    cout << "F\n";

  return 0;
}
