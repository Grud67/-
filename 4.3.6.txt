#include <iostream>
#include <cmath>
using namespace std;

int main() {
  double a, b, c;
  cout << "Введіть a, b, c: ";
  cin >> a >> b >> c;

  if (a == 0) {
    cout << "Це не квадратне рівняння (a = 0)\n";
    return 1;
  }

  double d = b * b - 4 * a * c;

  if (d > 0) {
    cout << "x1 = " << (-b + sqrt(d)) / (2 * a) << "\n";
    cout << "x2 = " << (-b - sqrt(d)) / (2 * a) << "\n";
  } else if (d == 0) {
    cout << "x1 = x2 = " << -b / (2 * a) << "\n";
  } else {
    cout << "Дійсних коренів немає (D < 0)\n";
  }

  return 0;
}
